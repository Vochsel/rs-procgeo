Cookbook

Complete workflow examples in Rust, JavaScript (WASM), and Python.

Subdivided Sphere

Create a low-poly sphere and smooth it with Catmull-Clark subdivision.

Rust

use procgeo::prelude::*;

let geo = generate(&SphereSop, &SphereParams {
    rows: 4,
    cols: 8,
    ..Default::default()
})?
.apply(&SubdivideSop, &SubdivideParams {
    depth: 2,
    mode: SubdivideMode::CatmullClark,
})?
.apply(&NormalSop, &NormalParams::default())?;

procgeo::io::write_file(&geo, Path::new("smooth_sphere.obj"))?;

JavaScript (WASM)

let geo = pg.createSphere({ radius: 0.8, rows: 4, cols: 8 });
geo = pg.subdivide(geo, { depth: 2, mode: 'catmullClark' });
geo = pg.computeNormals(geo);
return geo;

Python

import procgeo

geo = procgeo.create_sphere(radius=0.8, rows=4, cols=8)
geo = procgeo.subdivide(geo, depth=2)
geo = procgeo.compute_normals(geo)
procgeo.write_obj(geo, "smooth_sphere.obj")

Scatter and Instance

Scatter points on a grid, then copy a small box onto each point.

Rust

use procgeo::prelude::*;
use glam::Vec3;

// Create a grid to scatter on
let grid = generate(&GridSop, &GridParams {
    rows: 5, cols: 5,
    size: [4.0, 4.0],
    ..Default::default()
})?;

// Scatter 30 points on the grid surface
let points = grid.apply(&ScatterSop, &ScatterParams {
    count: 30, seed: 42,
})?;

// Create a small box to instance
let stamp = generate(&BoxSop, &BoxParams {
    size: Vec3::splat(0.15),
    ..Default::default()
})?;

// Copy box to each scattered point
let result = CopyToPointsSop.execute(
    &[&stamp, &points],
    &CopyToPointsParams::default(),
)?;

let result = result
    .apply(&NormalSop, &NormalParams::default())?
    .apply(&ColorSop, &ColorParams { color: [0.9, 0.5, 0.2] })?;

JavaScript (WASM)

const grid = pg.createGrid({ rows: 5, cols: 5, sizeX: 4, sizeY: 4 });
const points = pg.scatter(grid, { count: 30, seed: 42 });
const box = pg.createBox({ size: [0.15, 0.15, 0.15] });
let instances = pg.copyToPoints(box, points);
instances = pg.computeNormals(instances);
instances = pg.color(instances, { color: [0.9, 0.5, 0.2] });
return instances;

Python

import procgeo

grid = procgeo.create_grid(rows=5, cols=5, size_x=4.0, size_y=4.0)
points = procgeo.scatter(grid, count=30, seed=42)
box = procgeo.create_box(size_x=0.15, size_y=0.15, size_z=0.15)
instances = procgeo.copy_to_points(box, points)
instances = procgeo.compute_normals(instances)
instances = procgeo.color(instances, r=0.9, g=0.5, b=0.2)

Extruded City

Build a procedural cityscape by scattering extruded boxes.

Rust

use procgeo::prelude::*;
use glam::Vec3;

// Scatter building positions
let grid = generate(&GridSop, &GridParams {
    rows: 2, cols: 2,
    size: [8.0, 8.0],
    ..Default::default()
})?;
let sites = grid.apply(&ScatterSop, &ScatterParams { count: 12, seed: 7 })?;

// Create a building prototype
let building = generate(&BoxSop, &BoxParams {
    size: Vec3::new(0.8, 2.0, 0.8),
    ..Default::default()
})?
.apply(&PolyExtrudeSop, &PolyExtrudeParams {
    distance: 0.3,
    inset: 0.15,
    ..Default::default()
})?;

// Instance buildings at scattered points
let city = CopyToPointsSop.execute(
    &[&building, &sites],
    &CopyToPointsParams::default(),
)?
.apply(&NormalSop, &NormalParams::default())?
.apply(&ColorSop, &ColorParams { color: [0.6, 0.65, 0.75] })?;

JavaScript (WASM)

const grid = pg.createGrid({ rows: 2, cols: 2, sizeX: 8, sizeY: 8 });
const points = pg.scatter(grid, { count: 12, seed: 7 });
let building = pg.createBox({ size: [0.8, 2, 0.8] });
building = pg.polyExtrude(building, { distance: 0.3, inset: 0.15 });
let city = pg.copyToPoints(building, points);
city = pg.computeNormals(city);
city = pg.color(city, { color: [0.6, 0.65, 0.75] });
return city;

Voronoi Fracture

Fracture a box into Voronoi pieces for destruction effects.

Rust

use procgeo::prelude::*;
use glam::Vec3;

let box_geo = generate(&BoxSop, &BoxParams {
    size: Vec3::splat(1.5),
    ..Default::default()
})?;
let seeds = box_geo.apply(&ScatterSop, &ScatterParams { count: 6, seed: 42 })?;
let geo = VoronoiFractureSop.execute(
    &[&box_geo, &seeds],
    &VoronoiFractureParams::default(),
)?
.apply(&NormalSop, &NormalParams::default())?;

JavaScript (WASM)

let box = pg.createBox({ size: [1.5, 1.5, 1.5] });
let points = pg.scatter(box, { count: 6, seed: 42 });
let fractured = pg.voronoiFracture(box, points);
fractured = pg.computeNormals(fractured);
return fractured;

Working with Attributes

Create custom attributes and use them to drive geometry.

Rust

use procgeo::prelude::*;
use glam::Vec3;

let mut geo = generate(&GridSop, &GridParams {
    rows: 20, cols: 20,
    ..Default::default()
})?;

// Create a height attribute
geo.create_attrib::<[f32; 3]>(
    AttribClass::Point,
    "Cd",
    [1.0, 1.0, 1.0],
)?;

let handle = geo.find_attrib::<[f32; 3]>(
    AttribClass::Point,
    "Cd",
)?;

// Color each point based on height
for i in 0..geo.num_points() {
    let pos = geo.point_pos(PointHandle::from_index(i));
    let t = (pos.y + 5.0) / 10.0; // normalize to 0..1
    geo.set_attrib(&handle, i, [t, 0.3, 1.0 - t])?;
}

Working with Groups

Use groups to selectively delete parts of geometry.

Rust

use procgeo::prelude::*;

let mut geo = generate(&SphereSop, &SphereParams::default())?;

// Group all points above Y=0
geo.create_point_group("top_half");
{
    let group = geo.point_group_mut("top_half").unwrap();
    for i in 0..geo.num_points() {
        if geo.point_pos(PointHandle::from_index(i)).y > 0.0 {
            group.add(i);
        }
    }
}

// Remove the top half
let bottom = geo.apply(&BlastSop, &BlastParams {
    group_name: "top_half".into(),
    entity: BlastEntity::Points,
    ..Default::default()
})?;

Multi-Input SOPs

Some SOPs accept multiple inputs. Use .execute() directly for these.

Rust

use procgeo::prelude::*;
use glam::Vec3;

// Merge multiple geometries
let box1 = generate(&BoxSop, &BoxParams::default())?;
let box2 = generate(&BoxSop, &BoxParams {
    center: Vec3::new(2.0, 0.0, 0.0),
    ..Default::default()
})?;

let merged = MergeSop.execute(
    &[&box1, &box2],
    &MergeParams::default(),
)?;

Custom Feature Selection

Only compile the SOPs you need to minimize binary size.

# Cargo.toml — minimal build for mesh generation + OBJ export
[dependencies]
procgeo = { version = "0.1", default-features = false, features = [
    "creation",
    "transform",
    "normals",
    "reshape",
    "obj",
] }

Three.js Integration

Convert ProcGeo WASM geometry to Three.js BufferGeometry for rendering.

import * as THREE from 'three';

function toBufferGeometry(geo) {
    const bufGeo = new THREE.BufferGeometry();

    // Positions (flat Float32Array)
    bufGeo.setAttribute('position',
        new THREE.BufferAttribute(geo.getPositions(), 3));

    // Triangle indices
    bufGeo.setIndex(
        new THREE.BufferAttribute(geo.getTriangleIndices(), 1));

    // Normals (from "N" attribute if present)
    const normals = geo.getNormals?.();
    if (normals) {
        bufGeo.setAttribute('normal',
            new THREE.BufferAttribute(normals, 3));
    } else {
        bufGeo.computeVertexNormals();
    }

    // Vertex colors (from "Cd" attribute if present)
    const colors = geo.getColors?.();
    if (colors) {
        bufGeo.setAttribute('color',
            new THREE.BufferAttribute(colors, 3));
    }

    return bufGeo;
}

// Usage
const geo = pg.createSphere({ radius: 1 });
const withNormals = pg.computeNormals(geo);
const bufGeo = toBufferGeometry(withNormals);

const mesh = new THREE.Mesh(bufGeo, new THREE.MeshStandardMaterial({
    color: 0x4488cc,
    roughness: 0.55,
    metalness: 0.15,
}));
scene.add(mesh);
